#55 202004_A3-Escape (迷宮逃離)

題目摘要

問題敘述： 迷宮總共分為 N 個房間，每個房間的左側和右側各有一個雕像。玩家必須依照給定的 T 條線索，破壞每個房間中的其中一個雕像（即每個房間二選一），才能成功逃離迷宮。

每條線索包含兩個雕像位置（例如 1L、2R）以及一個關係字元：

  • S (Same)：代表兩個雕像的狀態相同（即同時保留或同時破壞）。
  • D (Different)：代表兩個雕像的狀態不同（即一個保留、一個破壞）。

已知所有線索，請判斷是否有辦法找到一種符合所有線索的雕像破壞方式。若存在則輸出 Easy Peasy，否則輸出 Impossible。

輸入格式： 第一行有兩個正整數 N 與 T (1 ≤ N ≤ 2000, 1 ≤ T ≤ N2)。接下來 T 行每行包含兩個雕像字串與一個關係字元 ('S' 或 'D')。

輸出格式： 若存在符合所有線索的破壞方式，輸出 Easy Peasy，否則輸出 Impossible。

演算法使用解析：【2-SAT (2-可滿足性問題) 與 Tarjan 強連通分量演算法】

本題可轉化為經典的 2-SAT 問題：

  • 變數定義： 對於每個房間 i (1 ≤ i ≤ N)，設布林變數 xi 代表房間 i 選擇破壞左側雕像 (iL)。因為每個房間必須破壞一個雕像，故選擇左側 (iL) 破壞即代表右側 (iR) 保留，反之亦然。因此 iL 與 iR 狀態互為否定。
  • 子句轉換 (Implications)：
    • 若線索為 A 與 B 狀態相同 (S)，代表 A ≤ftrightarrow B，可拆為 (A → B) ∧ (B → A) 以及 (≠g A → ≠g B) ∧ (≠g B → ≠g A)。
    • 若線索為 A 與 B 狀態不同 (D)，代表 A ≤ftrightarrow ≠g B，可拆為 (A → ≠g B) ∧ (B → ≠g A) 以及 (≠g A → B) ∧ (≠g B → A)。
  • 求解與判定： 透過建構蘊含圖 (Implication Graph) 並使用 Tarjan 演算法計算強連通分量 (SCC)。若任意房間變數 i 與其否定 ≠g i 落在同一個 SCC 中，代表發生矛盾，無解（輸出 Impossible）；否則代表有解（輸出 Easy Peasy）。

課堂黑板板書推導

【輸入範例 1 實際運作推導】

輸入：
3 2
1L 2L S
2L 3R S
  • N = 3 個房間，T = 2 條線索。
  • 線索 1：1L 2L S → 房間 1 左側與房間 2 左側狀態相同 (x1 ≤ftrightarrow x2)。
  • 線索 2：2L 3R S → 房間 2 左側與房間 3 右側狀態相同 (x2 ≤ftrightarrow ≠g x3)。
  • 透過 2-SAT 邊連接建立蘊含關係，執行 Tarjan 演算法檢查 SCC。無矛盾發生。

👉 輸出範例 1 結果： Easy Peasy

【輸入範例 2 實際運作推導】

輸入：
3 3
1L 2L S
2L 3R S
3L 1R D
  • N = 3，T = 3。
  • 加入線索 3：3L 1R D → 房間 3 左側與房間 1 右側狀態不同 (x3 ≤ftrightarrow ≠g (≠g x1) ⟺ x3 ≤ftrightarrow x1)。
  • 綜合三條線索會導致 x1 必須同時等於與不等於某個狀態，形成循環矛盾（變數與其否定在同一個強連通分量中）。

👉 輸出範例 2 結果： Impossible

參考解答 C++ 程式碼（含每行詳細中文教學註解）

#include <cstdio>
#include <vector>
#include <string>
#include <algorithm>
#include <stack>
using namespace std;

const int MAXN = 4005; // 2 * N (N <= 2000，每個房間有正負兩種狀態)

int n, t;
vector<int> adj[MAXN]; // 2-SAT 蘊含圖的鄰接串列
int dfn[MAXN], low[MAXN], timer; // Tarjan 演算法的時間戳記與追蹤變數
int scccnt, sccid[MAXN];      // 強連通分量計數與各節點所屬的 SCC 編號
bool instack[MAXN];            // 記錄節點是否目前在堆疊中
stack<int> st;                  // Tarjan 演算法專用堆疊

// 取得變數在 2-SAT 圖中的節點編號
// 變數 x (1 <= x <= n) 對應正 literal，若 x 為負代表否命題 (negation)
int getnode(int x) {
    if (x > 0) return 2 * x;       // 正 literal (例如 1L): 對應 2 * x
    else return 2 * (-x) + 1;      // 負 literal (例如 1R): 對應 2 * (-x) + 1
}

// 取得某個 literal 的否定 (negation)
int negatenode(int u) {
    return u ^ 1; // 偶數與奇數互為 XOR 1，可快速取得否定節點
}

// 加入蘊含關係 u -> v (等價於 ¬u ∨ v)
void addclause(int u, int v) {
    adj[u].pushback(v);
}

// Tarjan 演算法尋找強連通分量 (SCC)
void tarjan(int u) {
    dfn[u] = low[u] = ++timer;
    st.push(u);
    instack[u] = true;

    for (int v : adj[u]) {
        if (!dfn[v]) {
            tarjan(v);
            low[u] = min(low[u], low[v]);
        } else if (instack[v]) {
            low[u] = min(low[u], dfn[v]);
        }
    }

    if (dfn[u] == low[u]) {
        scccnt++;
        while (true) {
            int v = st.top();
            st.pop();
            instack[v] = false;
            sccid[v] = scccnt;
            if (u == v) break;
        }
    }
}

// 解析字串如 "1L" 或 "2R" 為 2-SAT 的 literal 編號
int parsestatue(const string& s) {
    int room = 0;
    int i = 0;
    while (i < s.size() && s[i] >= '0' && s[i] <= '9') {
        room = room * 10 + (s[i] - '0');
        i++;
    }
    char side = s[i]; // 'L' 或 'R'
    // 如果是 'L'，變數為 room；如果是 'R'，變數為 -room (代表選擇右側，即左側的否定)
    if (side == 'L') return getnode(room);
    else return getnode(-room);
}

void solve() {
    // 初始化圖形與變數
    for (int i = 0; i <= 2 * n + 1; ++i) {
        adj[i].clear();
        dfn[i] = low[i] = sccid[i] = 0;
        instack[i] = false;
    }
    timer = scccnt = 0;

    for (int i = 0; i < t; ++i) {
        char s1[10], s2[10], op[5];
        scanf("%s %s %s", s1, s2, op);
        int u = parsestatue(s1);
        int v = parsestatue(s2);

        if (op[0] == 'S') { // 狀態相同 (u <-> v)
            // (u -> v) 且 (v -> u) 以及 (¬u -> ¬v) 且 (¬v -> ¬u)
            addclause(u, v);
            addclause(negatenode(u), negatenode(v));
            addclause(v, u);
            addclause(negatenode(v), negatenode(u));
        } else { // 狀態不同 (u <-> ¬v)
            // (u -> ¬v) 且 (v -> ¬u) 以及 (¬u -> v) 且 (¬v -> u)
            addclause(u, negatenode(v));
            addclause(v, negatenode(u));
            addclause(negatenode(u), v);
            addclause(negatenode(v), u);
        }
    }

    // 執行 Tarjan 演算法尋找全圖的強連通分量
    for (int i = 2; i <= 2 * n + 1; ++i) {
        if (!dfn[i]) {
            tarjan(i);
        }
    }

    // 檢查是否有變數和其否定在同一個 SCC 中，若有則代表矛盾無解
    bool possible = true;
    for (int i = 1; i <= n; ++i) {
        int u = getnode(i);
        int negu = negatenode(u);
        if (sccid[u] == sccid[negu]) {
            possible = false;
            break;
        }
    }

    if (possible) printf("Easy Peasy\n");
    else printf("Impossible\n");
}

int main() {
    while (scanf("%d %d", &n, &t) != EOF) {
        solve(); // 處理每筆測資
    }
    return 0; // 程式正常結束
}
