#2 201903_A2-QueenBee (尋找蜂后)

PDF: 201903_A2-QueenBee(Q).pdf 開啟題目與原始解法 ↗

題目摘要

問題敘述： 科學家發現一種新型蜜蜂，其蜂巢結構為立方體，長、寬、高分別為 $L$、$W$、$H$。蜂巢被蜂巢壁分割為好幾個小空間（0 代表空間，1 代表蜂巢壁），蜂后住在蜂巢內體積最大（空格數最多）的連通空間中。請寫一個程式計算蜂巢中的最大空間包含多少格。

輸入格式： 第一列有三個正整數 $L, W, H$ ($1 \le L, W, H \le 135$)；第二列共有 $L \times W \times H$ 個字元，字元為 '0' 或 '1'，依序代表各層的蜂巢結構。

輸出格式： 輸出一列整數，代表蜂巢中的最大空間大小。

演算法使用解析：【廣度優先搜尋 (BFS) / 三維空間連通塊大小計算】

本題採用廣度優先搜尋 (BFS) 與三維網格走訪策略：

  • 三維空間走訪：每個點 $(x, y, z)$ 在三維空間中最多有 6 個相鄰方向（上、下、左、右、前、後）。
  • 連通塊計算：我們使用二維/三維陣列記錄拜訪狀態（visit）與佇列（queue）。當遇到尚未造訪的空白格 ('0') 時，即以其為起點發動 BFS，向 6 個方向擴散，將所有相連的空白格全部走訪過一次，並計算該連通區塊的大小。
  • 透過全域變數記錄並隨時更新最大空間（big = max(big, now)），最終即可找到蜂后所在的 最大空間。

課堂黑板板書推導

【輸入範例 1 詳細拆解與三維空間推導】

輸入：
3 3 3
010101010001001111111101111
  • 尺寸大小與結構： 長 $L = 3$，寬 $W = 3$，高 $H = 3$（總計 $3 \times 3 \times 3 = 27$ 個格子）。將一維字串依照高度分成 3 層來看：
    • 第 0 層 ($H=0$)： 0 1 0 / 1 0 1 / 0 1 0
    • 第 1 層 ($H=1$)： 0 0 0 / 1 0 0 / 1 1 1
    • 第 2 層 ($H=2$)： 1 1 1 / 1 1 0 / 1 1 1
  • BFS 擴散過程： 從任意未拜訪的空間 '0' 出發，透過 6 個方向（上下、左右、前後）在三維空間中向外擴散，走訪所有相連通的空白格。經過計算，此蜂巢中最大的連通空間總共包含 7 個格子的空間。

👉 輸出範例 1 結果： 7

【輸入範例 2 詳細拆解與三維空間推導】

輸入：
1 2 4
10000100
  • 尺寸大小與結構： 長 $L = 1$，寬 $W = 2$，高 $H = 4$（總計 $1 \times 2 \times 4 = 8$ 個格子）。
    • 第 0 層 ($H=0$)： 1 0
    • 第 1 層 ($H=1$)： 0 0
    • 第 2 層 ($H=2$)： 0 1
    • 第 3 層 ($H=3$)： 0 0
  • BFS 擴散過程： 排除蜂巢壁 '1' 後，其餘的 '0' 透過三維相鄰關係彼此連通，經過 BFS 計算，最大的連通空間大小為 6 格。

👉 輸出範例 2 結果： 6

參考解答 C++ 程式碼（百分之百保留原始解答邏輯與變數名稱，含每行詳細中文教學註解）

#include<bits/stdc++.h>  // 引入常用 C++ 標頭檔
using namespace std;

char arr[305][305][305]= {};  // 三維字元陣列，儲存蜂巢結構 ('0' 代表空間，'1' 代表蜂巢壁)
int visit[305][305][305]= {};  // 記錄三維空間中每個座標是否已經被 BFS 造訪過
int enqueue[305][305][305]= {};  // 記錄每個座標是否已經被加入過佇列中，避免重複進隊
int now=0;  // 記錄當前正在計算的連通空間大小
int big=0;  // 記錄所有連通空間中的最大空間大小 (即蜂后所在區域)

// 定義結構體儲存三維空間中的座標 (L, W, H)
struct st
{
    int L,W,H;
};

// BFS 函式：以廣度優先搜尋走訪並計算三維空間中連通的空白格數量
void BFS(st temp,int L,int W,int H)
{
    queue<st> next;  // 宣告儲存座標的標準函式庫佇列
    next.push(temp);  // 將起始點放入佇列中
    
    while(!next.empty())  // 當佇列不為空時持續擴散
    {
        st pos=next.front();  // 取出佇列最前端的座標
        now++;  // 當前連通空間大小加 1
        big=max(big,now);  // 更新全域最大空間紀錄
        next.pop();  // 將該點移出佇列
        
        int x=pos.H, y=pos.L, z=pos.W;  // 取得當前點的高(x)、長(y)、寬(z)座標
        enqueue[x][y][z]=0;  // 清除進隊標記
        visit[x][y][z]=1;  // 標記該點為已造訪
        
        // 定義三維空間中的 6 個移動方向（上下、左右、前後）
        int movex[6]={1,-1,0,0,0,0};
        int movey[6]={0,0,1,-1,0,0};
        int movez[6]={0,0,0,0,1,-1};
        
        // 迴圈檢查 6 個相鄰方向
        for(int i=0;i<6;i+=1)
        {
            int nx=x+movex[i];
            int ny=y+movey[i];
            int nz=z+movez[i];
            
            // 檢查新座標是否在蜂巢的合法範圍內
            if(nx>=0 && ny>=0 && nz>=0 && nx<H && ny<L && nz<W)
            {
                // 若該相鄰點未被造訪過、不是蜂巢壁('1')且不在佇列中
                if(!visit[nx][ny][nz] && arr[nx][ny][nz]!='1' && !enqueue[nx][ny][nz])
                {
                    next.push({ny,nz,nx});  // 將新座標推入佇列
                    enqueue[nx][ny][nz]=1;  // 標記已加入佇列
                }
            }
        }
    }
}

int main()
{
    int L,W,H;
    scanf("%d%d%d",&L,&W,&H);  // 讀入蜂巢的長 L、寬 W、高 H
    
    // 依序讀入三維蜂巢的每個字元狀態
    for(int i=0; i<H; i+=1)
    {
        for(int j=0; j<L; j+=1)
        {
            for(int k=0; k<W; k+=1)
            {
                scanf(" %c",&arr[i][j][k]);  // 讀入字元 ('0' 或 '1')
            }
        }
    }
    
    // 走訪三維蜂巢中的每一個格子，尋找未被造訪的空間起點
    for(int i=0; i<H; i+=1)
    {
        for(int j=0; j<L; j+=1)
        {
            for(int k=0; k<W; k+=1)
            {
                // 若該格是空間 ('0') 且尚未被造訪過
                if(arr[i][j][k]=='0' && !visit[i][j][k])
                {
                    now=0;  // 重置當前連通塊計數器
                    BFS({j,k,i},L,W,H);  // 發動 BFS 計算該連通空間大小
                }
            }
        }
    }
    
    printf("%d\n",big);  // 輸出整座蜂巢中的最大空間大小
}
