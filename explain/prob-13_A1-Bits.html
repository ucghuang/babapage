#13 201905_A1-Bits (位元加法)

題目摘要

問題敘述： 給定 $N$ 張二進位整數字卡（每張最多 15 個位元），以及一張代表目標數值的答案卡（$L$ 位元的二進位正整數， $1 \le L \le 19$）。請判斷是否能從這 $N$ 張字卡中挑選出其中幾張（至少一張），使其二進位加法總和剛好等於答案卡上的數字。

輸入格式： 第一列有一個正整數 $N$ ($1 \le N \le 100$)；接著 $N$ 列每列有一個二進位數字；最後一列有一個二進位正整數代表答案卡。

輸出格式： 若存在一組字卡總和等於答案卡則輸出 YES，否則輸出 NO。

演算法使用解析：【動態規劃 (DP) / 子集和概念與狀態記錄】

本題採用動態規劃與集合狀態記錄策略（類似背包問題的子集和問題）：

  • 首先將輸入的所有二進位字卡與答案卡利用自定義函式 bintodec 轉換為十進位整數，方便數值運算與比較。
  • 利用一個動態的向量 vector<int> v 與訪問記錄陣列 visit，依序遍歷每張字卡。對於每一張字卡，嘗試把它加到前面已經湊出來的所有可能組合中。
  • 如果過程中產生的總和剛好等於目標值（goal），則標記 found = true 並提早結束。透過布林陣列 visit 避免重複計算相同的組合總和，提升執行效率。

參考解答 C++ 程式碼（百分之百保留原始解答邏輯與變數名稱，含每行詳細中文教學註解）

#include<bits/stdc++.h>  // 引入常用 C++ 標頭檔
using namespace std;

// 自定義函式：將二進位字串轉換為十進位整數
int bintodec(char s[])
{
    int re=0;  // 儲存轉換後的十進位結果
    int base=1;  // 代表當前二進位位數的權重（從 2^0 開始）
    for(int i=strlen(s)-1; i>=0; i-=1)
    {
        re+=(base*(s[i]=='1'));  // 若該字元為 '1'，則加上當前權重
        base=base<<1;  // 權重每次往左移一位（乘以 2）
    }
    return re;  // 回傳十進位數值
}

bool visit[1200000]= {};  // 宣告全域布林陣列，用來記錄某個數字總和是否已經被湊出來過，避免重複

int main()
{
    int N;  // 宣告變數 N 代表字卡數量
    scanf("%d",&N);  // 讀入字卡數量 N
    int num[50005]= {};  // 宣告陣列儲存每張字卡的十進位數值
    for(int i=0; i<N; i+=1)
    {
        char s[25]= {};  // 宣告字元陣列讀取二進位字卡
        scanf("%s",s);  // 讀入二進位字卡字串
        num[i]=bintodec(s);  // 轉為十進位後存入陣列
    }
    char ans[30]= {};  // 宣告字元陣列讀取答案卡
    scanf("%s",ans);  // 讀入答案卡二進位字串
    int goal=bintodec(ans);  // 將答案卡轉為十進位目標值 goal
    
    vector<int> v;  // 宣告 vector 用來動態儲存所有可能湊出來的數字總和
    bool found=false;  // 旗標變數，記錄是否成功找到等於目標值的組合
    
    // 依序檢查每一張字卡
    for(int i=0; i<N; i+=1)
    {
        if(found) break;  // 若已經找到答案，直接跳出迴圈
        int len=v.size();  // 取得目前已經組合出來的總和數量
        
        // 將當前字卡與之前所有已組合出的總和相加
        for(int j=0; j<len; j+=1)
        {
            if(num[i]+v[j]>goal) continue;  // 若相加後超過目標值，則跳過不計
            else if(num[i]+v[j]==goal)  // 若相加後剛好等於目標值
            {
                found=true;  // 標記為找到解答
                break;  // 跳出內層迴圈
            }
            // 若新產生的總和尚未被訪問過
            if(!visit[num[i]+v[j]])
            {
                visit[num[i]+v[j]]=true;  // 標記為已訪問
                v.push_back(num[i]+v[j]);  // 加入到可能的總和集合中
            }
        }
        
        // 單獨檢查當前字卡本身
        if(!visit[num[i]] && num[i]<=goal)
        {
            visit[num[i]]=true;  // 標記已訪問
            v.push_back(num[i]);  // 加入集合
            if(num[i]==goal)  // 若單張字卡本身就等於目標值
            {
                found=true;  // 標記為找到解答
            }
        }
    }
    
    char print[2][20]={"NO","YES"};  // 定義輸出字串對應表 (索引 0 為 NO，索引 1 為 YES)
    printf("%s\n",print[found]);  // 根據 found 的布林值輸出對應結果
}
